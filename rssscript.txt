def nuke():
    # Switch to object mode if not already
    if bpy.context.active_object and bpy.context.active_object.mode != 'OBJECT':
        bpy.ops.object.mode_set(mode='OBJECT')

    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete(use_global=False)

    # Clean up orphaned data blocks
    for data_type in (
        bpy.data.meshes,
        bpy.data.materials,
        bpy.data.lights,
        bpy.data.cameras,
        bpy.data.images,
        bpy.data.textures
    ):
        for block in list(data_type):
            if block.users == 0:
                data_type.remove(block)

def mat(name, color, met=1.0, rough=.1, emit=None, strength=1):
    m = bpy.data.materials.new(name)
    m.use_nodes = True
    m.diffuse_color = color # Set viewport color
    
    nodes = m.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    
    if not bsdf:
        bsdf = nodes.new(type='ShaderNodeBsdfPrincipled')
    
    # Ensure BSDF is connected to Output
    output = nodes.get("Material Output")
    if not output:
        output = nodes.new(type='ShaderNodeOutputMaterial')
    
    # Clear existing links to avoid mess and create a clean connection
    m.node_tree.links.clear()
    m.node_tree.links.new(bsdf.outputs['BSDF'], output.inputs['Surface'])
        
    def set_input(node, name, value):
        if name in node.inputs:
            node.inputs[name].default_value = value
        # Special case for Blender 4.0+ Emission
        elif name == "Emission Color" and "Emission" in node.inputs:
             node.inputs["Emission"].default_value = value

    set_input(bsdf, "Base Color", color)
    set_input(bsdf, "Metallic", met)
    set_input(bsdf, "Roughness", rough)

    if emit:
        # In 4.0+ 'Emission Color' is the label but internally it's often 'Emission'
        # or vice versa. We try both to be safe.
        if "Emission Color" in bsdf.inputs:
            bsdf.inputs["Emission Color"].default_value = emit
        elif "Emission" in bsdf.inputs:
            bsdf.inputs["Emission"].default_value = emit
            
        if "Emission Strength" in bsdf.inputs:
            bsdf.inputs["Emission Strength"].default_value = strength

    return m

def parent_obj(c,p):
    c.parent=p
    c.matrix_parent_inverse=p.matrix_world.inverted()

def key(o, f, l=None, r=None, s=None, emit_strength=None):

    if l:
        o.location = l
        o.keyframe_insert("location", frame=f)

    if r:
        o.rotation_euler = r
        o.keyframe_insert("rotation_euler", frame=f)

    if s:
        o.scale = s
        o.keyframe_insert("scale", frame=f)
        
    if emit_strength is not None and o.data and hasattr(o.data, "materials") and o.data.materials:
        mat = o.data.materials[0]
        if mat.use_nodes:
            bsdf = mat.node_tree.nodes.get("Principled BSDF")
            if bsdf and "Emission Strength" in bsdf.inputs:
                bsdf.inputs["Emission Strength"].default_value = emit_strength
                bsdf.inputs["Emission Strength"].keyframe_insert("default_value", frame=f)

def import_tool_obj(filepath, name, materials, scale, rot_x=0, rot_y=0, rot_z=0):
    """
    Imports OBJ and applies a mixed nanotech material look.
    materials: a list/tuple of materials [primary, secondary, accent]
    """
    if not os.path.exists(filepath):
        print(f"MISSING: {filepath}")
        return []

    before = set(bpy.data.objects.keys())
    try:
        bpy.ops.wm.obj_import(filepath=filepath)
    except:
        try:
            bpy.ops.import_scene.obj(filepath=filepath)
        except:
            return []

    after = set(bpy.data.objects.keys())
    new_objs = [bpy.data.objects[n] for n in (after - before)]

    bpy.ops.object.select_all(action='DESELECT')
    for o in new_objs:
        o.select_set(True)

    def apply_nanotech_mats(o):
        o.data.materials.clear()
        for m in materials:
            o.data.materials.append(m)
        
        if hasattr(o.data, "polygons"):
            for i, p in enumerate(o.data.polygons):
                # Logic for a nanotech "mixed" look:
                # 70% primary, 25% secondary, 5% accent sparks
                seed = (i * 13) % 100
                if seed < 70:
                    p.material_index = 0 # Primary
                elif seed < 95:
                    p.material_index = 1 # Secondary
                else:
                    p.material_index = 2 # Accent

    for o in new_objs:
        apply_nanotech_mats(o)

    bpy.context.view_layer.objects.active = new_objs[0]
    if len(new_objs) > 1:
        bpy.ops.object.join()

    obj = bpy.context.active_object
    obj.name = name
    
    # Re-apply distribution after joining
    apply_nanotech_mats(obj)

    bpy.ops.object.origin_set(type='ORIGIN_GEOMETRY', center='BOUNDS')
    obj.location = (0, 0, 0)
    obj.rotation_euler = (rot_x, rot_y, rot_z)
    obj.scale = (scale, scale, scale)
    bpy.ops.object.transform_apply(location=True, rotation=True, scale=True)

    obj["orig_loc"] = Vector(obj.location)
    bpy.ops.object.shade_smooth()
    return [obj]

def animate_morph(f_start, f_end, active_parts=None, tool_z=0.5, scale_multiplier=1.0):
    """
    Ultimate Cinematic Transformation:
    1. Ignite (f_start): Arc flares, energy surge.
    2. Flow (f_flow): Shards explode out in organic paths.
    3. Assemble (f_assemble): Complex vortex assembly with staggered parts.
    4. Impact (f_end): Tool locks, camera shake, energy snap.
    """
    duration = f_end - f_start
    f_ignite = f_start + (duration * 0.1)
    f_flow = f_start + (duration * 0.4)
    f_assemble = f_start + (duration * 0.75)
    
    # 1. Handle tools scaling and location (STAGGERED BUILD)
    for i, o in enumerate(all_tools):
        target_loc = Vector(o.get("orig_loc", (0, 0, 0)))
        
        # Stagger based on object index for "part-by-part" assembly
        stagger = (i % 5) * 3 
        f_s_start = f_start + stagger
        f_s_flow = f_flow + stagger
        f_s_assemble = f_assemble + stagger

        if active_parts and o in active_parts:
            # MATERIALIZE
            key(o, f_start, s=(0.001, 0.001, 0.001), l=(0, 0, 0), r=(0, 0, 0))
            # Heat glow/energy surge during assembly - adding more vibration
            vib = 0.05
            key(o, f_s_flow, s=(0.1 * scale_multiplier, 0.1 * scale_multiplier, 0.1 * scale_multiplier), l=(random.uniform(-vib,vib), random.uniform(-vib,vib), target_loc.z * 0.2), r=(0, 0, math.pi), emit_strength=400)
            key(o, f_s_assemble, s=(0.95 * scale_multiplier, 0.95 * scale_multiplier, 0.95 * scale_multiplier), l=(random.uniform(-vib/2,vib/2), random.uniform(-vib/2,vib/2), target_loc.z * 0.95), r=(0, 0, math.pi*1.8), emit_strength=200)
            # Final lock with a little overshoot
            key(o, f_end - 2, s=(1.05 * scale_multiplier, 1.05 * scale_multiplier, 1.05 * scale_multiplier), l=target_loc, r=(0, 0, 0), emit_strength=50)
            key(o, f_end, s=(scale_multiplier, scale_multiplier, scale_multiplier), l=target_loc, r=(0, 0, 0), emit_strength=1)
        else:
            # DE-MATERIALIZE
            if f_start == 1:
                key(o, 1, s=(0.001, 0.001, 0.001), l=(0, 0, 0))
            else:
                if o.scale.x > 0.1:
                    key(o, f_ignite, s=(1, 1, 1), l=target_loc, r=(0, 0, 0))
                    key(o, f_flow, s=(0.4, 0.4, 0.4), l=(0, 0, target_loc.z * 0.4), r=(0, 0, math.pi*0.5), emit_strength=300)
                    key(o, f_end, s=(0.001, 0.001, 0.001), l=(0, 0, 0), r=(0, 0, math.pi), emit_strength=1)

    # 2. Handle Swarm (The Nanobots)
    for i, p in enumerate(swarm):
        seed = i * 3.14159
        p_type = p.get("type", "core")
        a = math.radians(i * (360/len(swarm)))
        
        # Turbulence
        tx, ty, tz = math.sin(seed), math.cos(seed), math.sin(seed*2.5)
        
        if active_parts:
            # FLOW (Explosion Phase) - tighter paths
            dist = 0.8 if p_type == "spark" else 0.4
            key(p, f_flow, 
                l=(math.cos(a)*dist + tx*0.3, math.sin(a)*dist + ty*0.3, tool_z*0.5 + tz*0.5),
                r=(tx*20, ty*20, tz*20),
                s=(2.0, 2.0, 2.0)
            )
            
            # ASSEMBLE (TIGHT VORTEX COMPRESSION)
            # Spiral count increases for tighter compression
            a_vort = a + (math.pi*15 if p_type == "spark" else math.pi*8)
            rad = 0.03 if p_type == "core" else 0.1
            key(p, f_assemble, 
                l=(math.cos(a_vort)*rad, math.sin(a_vort)*rad, tool_z + tz*0.03),
                r=(a_vort*2, a_vort*2, a_vort*2),
                s=(0.8, 0.8, 0.8)
            )
            
            # IMPACT (Absorption)
            key(p, f_end, l=(0, 0, tool_z), s=(0, 0, 0))
        else:
            # RETRACT - more aggressive spiral back
            a_ret = a - (math.pi * 5)
            key(p, f_flow, l=(math.cos(a_ret)*0.4, math.sin(a_ret)*0.4, -0.05), s=(1.8, 1.8, 1.8))
            key(p, f_end, l=(math.cos(a)*0.16 + tx*0.01, math.sin(a)*0.16 + ty*0.01, -0.25 + tz*0.01), r=(0, 0, a), s=(1.0, 1.0, 1.0))

    # 3. Camera Shake and FX
    cam = scene.camera
    if cam:
        key(cam, f_start, l=(0, -20, 3))
        # More aggressive shake on impact
        key(cam, f_end - 3, l=(0.12, -20.12, 3.12))
        key(cam, f_end - 1, l=(-0.08, -19.92, 2.92))
        key(cam, f_end, l=(0, -20, 3))

    # 4. Arc Flare (Extreme surge)
    key(h_arc, f_start, s=(1, 1, 1), emit_strength=50)
    key(h_arc, f_ignite, s=(2.2, 2.2, 2.2), emit_strength=600) # Maximum flare
    key(h_arc, f_flow, s=(1.5, 1.5, 1.5), emit_strength=250)
    key(h_arc, f_assemble, s=(1.2, 1.2, 1.2), emit_strength=150)
    key(h_arc, f_end, s=(1, 1, 1), emit_strength=50)

class JARVIS_OT_Listener(bpy.types.Operator):
    """Ultimate J.A.R.V.I.S. Listener with Heartbeat Safety"""
    bl_idname = "wm.jarvis_listener"
    bl_label = "J.A.R.V.I.S. Listener"
    
    _timer = None
    last_active_parts = None
    current_scale = 1.0
    last_command = None  # Track last command to prevent automatic re-execution
    
    def should_execute_command(self, cmd):
        """Check if command should execute - only NEW commands allowed"""
        if cmd == self.last_command:
            # Same command - ignore to prevent automatic re-conversion
            return False
        
        # Transform commands (BIGGER, SMALLER, BREAK, REPAIR) require explicit fresh tool
        transform_cmds = ["BIGGER", "SMALLER", "BREAK", "REPAIR", "REGENERATE"]
        if cmd in transform_cmds:
            # For transform commands, check if we have an active tool AND this is a NEW command
            if self.last_active_parts is None:
                print(f"J.A.R.V.I.S.: Cannot {cmd} - no tool selected. Please select a tool first.")
                return False
            # Command is new and we have a tool - allow it
            return True
        
        # Tool selection commands always execute when new
        return True
    
    def modal(self, context, event):
        if event.type == 'TIMER':
            # 1. HEARTBEAT CHECK (Is J.A.R.V.I.S. AI running?)
            is_ai_alive = False
            if os.path.exists(HEARTBEAT_FILE):
                try:
                    with open(HEARTBEAT_FILE, "r") as f:
                        last_heartbeat = float(f.read().strip())
                    time_diff = time.time() - last_heartbeat
                    if time_diff < 30: # Increased to 30s for safety
                        is_ai_alive = True
                    else:
                        if int(time.time()) % 5 == 0:
                            print(f"J.A.R.V.I.S. Status: Heartbeat too old ({time_diff:.1f}s)")
                except Exception as e:
                    print(f"Heartbeat Read Error: {e}")
            else:
                if int(time.time()) % 5 == 0:
                    print("J.A.R.V.I.S. Status: Heartbeat file missing")
            
            if not is_ai_alive:
                # DO NOT remove the command file here, just wait
                return {'PASS_THROUGH'}

            # 2. COMMAND CHECK
            if os.path.exists(COMMAND_FILE):
                try:
                    with open(COMMAND_FILE, "r") as f:
                        cmd = f.read().strip().upper()
                    
                    if cmd:
                        print(f"J.A.R.V.I.S. Bridge: Received '{cmd}'")
                        
                        # CHECK: Should we execute this command?
                        if not self.should_execute_command(cmd):
                            if cmd == self.last_command:
                                print(f"J.A.R.V.I.S.: Ignoring duplicate '{cmd}' - waiting for NEW command")
                            # Remove command file to prevent re-processing
                            try:
                                os.remove(COMMAND_FILE)
                            except:
                                pass
                            return {'PASS_THROUGH'}
                        
                        # Store this as the last executed command
                        self.last_command = cmd
                        print(f"J.A.R.V.I.S.: Command '{cmd}' accepted - executing...")
                        
                        # Remove command file AFTER processing to prevent race conditions
                        # os.remove(COMMAND_FILE)
                        
                        # Ensure we are in a valid state for animation
                        f_start = context.scene.frame_current
                        f_end = f_start + 30
                        
                        # Set scene end frame if needed to allow animation
                        if context.scene.frame_end < f_end + 10:
                            context.scene.frame_end = f_end + 1000
                            
                        # 1. TRANSFORM COMMANDS
                        if cmd == "HAMMER":
                            print("J.A.R.V.I.S.: Deploying Hammer...")
                            self.last_active_parts = hammer_parts
                            animate_morph(f_start, f_end, hammer_parts, 0.1, self.current_scale)

    def execute(self, context):
        wm = context.window_manager
        self._timer = wm.event_timer_add(0.1, window=context.window)  # Faster polling
        wm.modal_handler_add(self)
        print("J.A.R.V.I.S. Listener Started - Checking for commands every 0.1s")
        print(f"Command file path: {COMMAND_FILE}")
        print(f"Heartbeat file path: {HEARTBEAT_FILE}")
        return {'RUNNING_MODAL'}

    def cancel(self, context):
        wm = context.window_manager
        wm.event_timer_remove(self._timer)
        print("J.A.R.V.I.S. Listener Stopped.")
